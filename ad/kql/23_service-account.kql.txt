// Sanitized synthetic Active Directory hunt
// No client, tenant, or production identifiers.
DeviceLogonEvents
| where Timestamp >= ago(14d) and AccountName startswith "svc-"
| summarize Hosts=make_set(DeviceName, 30), HostCount=dcount(DeviceName), Interactive=countif(LogonType in ("Interactive", "RemoteInteractive")) by AccountName, RemoteIP, bin(Timestamp, 1h)
| where HostCount >= 3 or Interactive > 0
