// Sanitized synthetic Active Directory hunt
// No client, tenant, or production identifiers.
SecurityEvent
| where TimeGenerated >= ago(14d) and EventID == 4662
| where Properties has_any ("1131f6aa-9c07-11d1-f79f-00c04fc2dcd2", "1131f6ad-9c07-11d1-f79f-00c04fc2dcd2")
| where SubjectUserName !endswith "$"
| summarize Operations=count(), Objects=make_set(ObjectName, 20) by SubjectUserName, IpAddress, Computer
