// Sanitized synthetic Active Directory hunt
// No client, tenant, or production identifiers.
SecurityEvent
| where TimeGenerated >= ago(24h) and EventID == 4625
| where Status in ("0xC000006A", "0xC000006D")
| summarize Failures=count(), TargetUsers=dcount(TargetUserName), Users=make_set(TargetUserName, 50) by IpAddress, Computer, bin(TimeGenerated, 10m)
| where TargetUsers >= 8 and Failures >= 10
