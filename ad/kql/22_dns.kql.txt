// Sanitized synthetic Active Directory hunt
// No client, tenant, or production identifiers.
SecurityEvent
| where TimeGenerated >= ago(30d) and EventID in (5136, 5137)
| where EventData has_any ("dnsNode", "dnsRecord", "DC=DomainDnsZones")
| project TimeGenerated, Computer, SubjectUserName, EventID, ObjectDN, EventData
