T1021.002 · Remote Services: SMB/Windows Admin Shares

SMB can be used to reach hidden administrative shares such as C$ and ADMIN$, transfer files, and support remote execution. This lab implements the multi-event idea: share access first, then service, task, file, or process evidence.

Open ATT&CK technique ↗
Valid account→SMB / 445→Admin share→Target action
Simple decision rule
  1. Scope: source and target are workstations.
  2. Access: share is exactly ADMIN$ or C$.
  3. Correlate: same source, target, and account.
  4. Time: follow-on event is 0–300 seconds later.
  5. Tune: preserve exact management-source allowlists.

Answer panel
KQL
Live tenant validation required
let Window = 5m;
let ManagementSources = dynamic(["SCCM-01", "PATCH-01", "BACKUP-01", "VSCAN-01", "HD-WS-01"]);
let Shares = SecurityEventNormalized
| where EventType == "share_access"
| where ShareName in~ ("ADMIN$", "C$")
| where SourceType =~ "workstation" and TargetType =~ "workstation"
| where SourceDevice !in~ (ManagementSources)
| project ShareTime=TimeGenerated, SourceDevice, TargetDevice, Account, ShareName, SharePath=Path;
let FollowOn = SecurityEventNormalized
| where EventType in~ ("file_write", "service_install", "process_start")
| project FollowTime=TimeGenerated, SourceDevice, TargetDevice, Account, EventType, Path, ProcessName, ServiceName;
Shares
| join kind=inner FollowOn on SourceDevice, TargetDevice, Account
| where FollowTime between (ShareTime .. ShareTime + Window)
| summarize FirstShare=min(ShareTime), LastFollowOn=max(FollowTime),
            FollowOnEvents=make_set(EventType), Paths=make_set(Path),
            Processes=make_set(ProcessName), Services=make_set(ServiceName)
  by SourceDevice, TargetDevice, Account, ShareName, SharePath
| where array_length(FollowOnEvents) > 0
| order by FirstShare desc


Analyst reasoning
Why this needs tuning
The same chain can be produced by software deployment, patching, backup, vulnerability scanning, and help-desk work. Prefer exact approved source devices, service accounts, maintenance windows, and ticket evidence. Avoid excluding every event from a management VLAN because a compromised workstation could be moved into that range.
Safe narrow filterSourceDevice == "SCCM-01"
Risky broad filterSourceIP startswith "10.20."

Answer panel
Python
Executed against all fixtures
import json
from datetime import datetime

WINDOW_SECONDS = 300
ALLOWLIST = {"SCCM-01", "PATCH-01", "BACKUP-01", "VSCAN-01", "HD-WS-01"}
FOLLOW_ON = {"file_write", "service_install", "process_start"}

def time_value(text):
    return datetime.fromisoformat(text.replace("Z", "+00:00"))

with open("week16_events.jsonl", encoding="utf-8") as handle:
    events = [json.loads(line) for line in handle if line.strip()]

shares = [event for event in events
          if event["event_type"] == "share_access"
          and event.get("share", "").upper() in {"ADMIN$", "C$"}
          and event.get("source_type") == "workstation"
          and event.get("target_type") == "workstation"
          and event["source_device"] not in ALLOWLIST]

for share in shares:
    matches = []
    for event in events:
        same_route = (event["source_device"], event["target_device"], event["account"]) == (
            share["source_device"], share["target_device"], share["account"])
        seconds = (time_value(event["timestamp"]) - time_value(share["timestamp"])).total_seconds()
        if same_route and event["event_type"] in FOLLOW_ON and 0 <= seconds <= WINDOW_SECONDS:
            matches.append(event)
    if matches:
        print(f'{share["scenario"]}: {share["source_device"]} -> {share["target_device"]}')
        for event in matches:
            print(f'  +{int((time_value(event["timestamp"]) - time_value(share["timestamp"])).total_seconds())}s {event["event_type"]}: {event.get("path", "-")}')

package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "sort"
    "strings"
)

type Event struct {
    EventType   string `json:"event_type"`
    Source      string `json:"source_device"`
    Target      string `json:"target_device"`
    SourceType  string `json:"source_type"`
    TargetType  string `json:"target_type"`
    Share       string `json:"share"`
}

type Edge struct {
    Source string
    Target string
    Count  int
}

func main() {
    file, err := os.Open("week16_events.jsonl")
    if err != nil { panic(err) }
    defer file.Close()

    allowed := map[string]bool{"SCCM-01": true, "PATCH-01": true, "BACKUP-01": true, "VSCAN-01": true, "HD-WS-01": true}
    counts := map[string]int{}
    scanner := bufio.NewScanner(file)

    for scanner.Scan() {
        var event Event
        if json.Unmarshal(scanner.Bytes(), &event) != nil { continue }
        adminShare := event.Share == "ADMIN$" || event.Share == "C$"
        workstationPair := event.SourceType == "workstation" && event.TargetType == "workstation"
        if event.EventType == "share_access" && adminShare && workstationPair && !allowed[event.Source] {
            counts[event.Source+"|"+event.Target]++
        }
    }
    if err := scanner.Err(); err != nil { panic(err) }

    edges := []Edge{}
    for key, count := range counts {
        parts := strings.SplitN(key, "|", 2)
        edges = append(edges, Edge{parts[0], parts[1], count})
    }
    sort.Slice(edges, func(i, j int) bool {
        if edges[i].Source == edges[j].Source { return edges[i].Target < edges[j].Target }
        return edges[i].Source < edges[j].Source
    })

    fmt.Println("SOURCE -> TARGET | ADMIN-SHARE EVENTS")
    for _, edge := range edges {
        fmt.Printf("%s -> %s | %d\n", edge.Source, edge.Target, edge.Count)
    }
}

Analyst reasoning
False positives to verify
  • Software deployment: approved management source, package path, service account, and change.
  • Patching: maintenance window, approved patch runner, and expected payload.
  • Backup: read-heavy activity from known backup infrastructure.
  • Vulnerability scanning: scheduled scan ID and expected service-query behavior.
  • Help desk: exact support workstation, assigned technician, and active ticket.
Investigation pivots
Before escalation
  1. Confirm source and target device roles.
  2. Check whether the account should administer the target.
  3. Hash and signer-review the transferred file.
  4. Inspect service, scheduled-task, and process telemetry.
  5. Look for fan-out from the source to more workstations.
  6. Validate the allowlist owner and expiry date.
