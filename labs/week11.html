KQL behavior query + tuning

Page 3 / 8

What should be done

  1. Start with service-install events.
  2. Normalize fields from AdditionalFields.
  3. Extract the executable from quoted or unquoted image paths.
  4. Keep the four risky path families.
  5. Remove only approved publisher + service + time combinations.

Why tuning is narrow

A signer alone is not enough. A stolen certificate, renamed service, or unexpected install time must remain visible. The example exception therefore requires the approved signer, the expected service, and the deployment window.

Replace the sample publishers, service names, and UTC window with change-controlled values from your environment.

KQL answer

Starting point for Microsoft Defender Advanced Hunting. Confirm your tenant's action type and AdditionalFields names with one known service event.

let WindowStart = datetime(2026-11-10T02:00:00Z);
let WindowEnd = datetime(2026-11-10T04:00:00Z);
DeviceEvents
| where ActionType in~ ("ServiceInstalled", "ServiceCreated")
| extend ServiceName = tostring(AdditionalFields.ServiceName),
         ImagePath = tostring(AdditionalFields.ImagePath),
         Actor = coalesce(InitiatingProcessAccountName,
                          tostring(AdditionalFields.Actor)),
         Signer = tostring(AdditionalFields.Signer),
         SHA256 = tostring(AdditionalFields.SHA256)
| extend BinaryPath = coalesce(
    extract(@'^"([^"]+\.exe)"', 1, ImagePath),
    extract(@'^(.+?\.exe)(?:\s|$)', 1, ImagePath)
  )
| extend LowerPath = tolower(BinaryPath)
| extend RiskyPath =
    LowerPath startswith @"\\"
    or LowerPath contains @"\appdata\"
    or LowerPath contains @"\temp\"
    or LowerPath startswith @"c:\programdata\"
| where RiskyPath
| extend ApprovedDeployment =
    (Signer =~ "Contoso Software Ltd." and ServiceName =~ "ContosoDeployAgent"
      and Timestamp between (WindowStart .. WindowEnd))
    or
    (Signer =~ "SecureShield Inc." and ServiceName =~ "SafeBackupAgent"
      and Timestamp between (WindowStart .. WindowEnd))
| where not(ApprovedDeployment)
| project Timestamp, DeviceName, ServiceName, BinaryPath,
          Actor, Signer, SHA256
| order by Timestamp desc

Analyst reasoning

Likely false positives: EDR or backup agents installed from a share, legitimate installers that unpack into Temp, and managed deployments that stage binaries in ProgramData. Verify the publisher, certificate chain, service name, binary hash, software owner, change ticket, target devices, and deployment window before tuning. Do not allowlist an entire directory or signer.

Week 11 · Day 5

Python path extraction + scoring

Page 6 / 8

What should be done

  1. Read one JSON object per line.
  2. Extract the executable from a quoted or unquoted image path.
  3. Add simple points for share, AppData, Temp, or ProgramData.
  4. Add points when the signer is unsigned or unknown.
  5. Skip only the exact approved deployments.

Score guide

40UNC share
35AppData
30Temp

ProgramData adds 20; unsigned/unknown signer adds 15. Scores prioritize review—they do not prove malicious intent.

Python answer

import json
import re
from datetime import datetime, timezone

WINDOW_START = datetime(2026, 11, 10, 2, 0, tzinfo=timezone.utc)
WINDOW_END = datetime(2026, 11, 10, 4, 0, tzinfo=timezone.utc)
APPROVED = {
    ("contosodeployagent", "contoso software ltd."),
    ("safebackupagent", "secureshield inc."),
}


def binary_path(image_path):
    match = re.match(r'^"([^"]+\.exe)"|^(.+?\.exe)(?:\s|$)', image_path, re.I)
    return (match.group(1) or match.group(2)) if match else image_path


def in_window(timestamp):
    value = datetime.fromisoformat(timestamp.replace("Z", "+00:00"))
    return WINDOW_START <= value <= WINDOW_END


def approved(event):
    pair = (event["service_name"].lower(), event["signer"].lower())
    return pair in APPROVED and in_window(event["timestamp"])


def score(event):
    path = binary_path(event["image_path"]).lower()
    points, reasons = 0, []
    if path.startswith("\\\\"):
        points += 40
        reasons.append("share")
    if "\\appdata\\" in path:
        points += 35
        reasons.append("AppData")
    if "\\temp\\" in path:
        points += 30
        reasons.append("Temp")
    if path.startswith("c:\\programdata\\"):
        points += 20
        reasons.append("ProgramData")
    if event["signer"].lower() in {"unsigned", "unknown"}:
        points += 15
        reasons.append("untrusted signer")
    return points, reasons


with open("events.jsonl", encoding="utf-8") as source:
    for line in source:
        event = json.loads(line)
        points, reasons = score(event)
        if points > 0 and not approved(event):
            path = binary_path(event["image_path"])
            print(f'{event["service_name"]}: {points} | {path} | {", ".join(reasons)}')

Go answer

package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "regexp"
    "sort"
    "strings"
)

type Event struct {
    ServiceName string `json:"service_name"`
    ImagePath   string `json:"image_path"`
    Signer      string `json:"signer"`
}

var exe = regexp.MustCompile(`(?i)^"([^"]+\.exe)"|^(.+?\.exe)(?:\s|$)`)

func binaryPath(value string) string {
    match := exe.FindStringSubmatch(value)
    if len(match) == 0 {
        return value
    }
    if match[1] != "" {
        return match[1]
    }
    return match[2]
}

func bucket(path string) string {
    lower := strings.ToLower(path)
    switch {
    case strings.HasPrefix(lower, `\\`):
        return "Share"
    case strings.Contains(lower, `\appdata\`):
        return "AppData"
    case strings.Contains(lower, `\temp\`):
        return "Temp"
    case strings.HasPrefix(lower, `c:\programdata\`):
        return "ProgramData"
    default:
        return "Other"
    }
}

func main() {
    file, err := os.Open("events.jsonl")
    if err != nil {
        panic(err)
    }
    defer file.Close()

    counts := map[string]int{}
    scanner := bufio.NewScanner(file)
    for scanner.Scan() {
        var event Event
        if err := json.Unmarshal(scanner.Bytes(), &event); err != nil {
            panic(err)
        }
        key := fmt.Sprintf("%-24s %-12s", event.Signer, bucket(binaryPath(event.ImagePath)))
        counts[key]++
    }
    if err := scanner.Err(); err != nil {
        panic(err)
    }

    keys := make([]string, 0, len(counts))
    for key := range counts {
        keys = append(keys, key)
    }
    sort.Strings(keys)
    for _, key := range keys {
        fmt.Printf("%s %d\n", key, counts[key])
    }
}

Week 11 · Day 7

Capstone: correlate and tune

Page 8 / 8

Suspicious three-event sequence

Correlate on the same device and binary path or SHA-256. Require the file drop to occur before service creation, and the service start to occur after creation, within ten minutes.

Decision path

Risky service path?No: keep quiet
→
Drop + install + start?Yes: raise confidence
→
Approved exact deployment?Yes: suppress and record

Tuning guardrails

  • Approved publisher and valid certificate
  • Expected service name and binary hash/path
  • Named software owner and change ticket
  • Exact device scope
  • UTC start and end of deployment window
  • Expiry and review date for every exception

Correlation KQL

Adapt the normalized action and field names after inspecting your tenant. The joins deliberately keep a short ten-minute sequence.

let Drops = DeviceFileEvents
| project DeviceId, DropTime=Timestamp, BinaryPath=FolderPath,
          SHA256, DropActor=InitiatingProcessAccountName;
let Installs = DeviceEvents
| where ActionType in~ ("ServiceInstalled", "ServiceCreated")
| extend ServiceName=tostring(AdditionalFields.ServiceName),
         BinaryPath=tostring(AdditionalFields.BinaryPath),
         SHA256=tostring(AdditionalFields.SHA256),
         Signer=tostring(AdditionalFields.Signer)
| project DeviceId, InstallTime=Timestamp, ServiceName,
          BinaryPath, SHA256, Signer;
let Starts = DeviceProcessEvents
| where InitiatingProcessFileName =~ "services.exe"
| project DeviceId, StartTime=Timestamp, BinaryPath=FolderPath, SHA256;
Drops
| join kind=inner Installs on DeviceId, BinaryPath, SHA256
| where InstallTime between (DropTime .. DropTime + 10m)
| join kind=inner Starts on DeviceId, BinaryPath, SHA256
| where StartTime between (InstallTime .. InstallTime + 10m)
| project DeviceId, ServiceName, BinaryPath, SHA256, Signer,
          DropActor, DropTime, InstallTime, StartTime

Analyst reasoning: false positives

EDR / backup agents

These tools may install services from shares. Confirm a trusted publisher, expected service name, central deployment record, scoped devices, and a narrow window.

Legitimate installers

Installers often unpack to Temp before registering a service. Verify the parent installer, signature chain, package hash, software owner, and uninstall metadata.

Temporary deployment paths

ProgramData staging can be normal. Require a change ticket and expire the exception immediately after deployment; an out-of-window install must alert.

Honest limits

  • KQL was not executed in a live Microsoft Defender tenant.
  • CrowdStrike CQL was not executed in a live Falcon tenant.
  • Event names and normalized fields must be confirmed locally.
  • The YARA rule is only for the harmless fixture, not production service detection.
