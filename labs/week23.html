Week 23 · Day 2

KQL behavior query + tuning

Page 3 / 8

What should be done

  1. Make a creation table with key, creator, source, role, and workflow.
  2. Make a use table with the earliest successful API call per key.
  3. Join on AccessKeyId.
  4. Keep 0–15 minute deltas and changed sources.
  5. Filter only approved role/workflow pairs.

Tuning guardrails

Role alone is too broad. This lab requires both the expected creator role and expected workflow name. Keep an owner, ticket, expiry, and known runner range with every production allowlist entry.

Do not allowlist all access-key rotation or all administrator activity.

KQL answer

Validation place: Microsoft Sentinel Logs after confirming the AWS CloudTrail connector's table and column names.

let Window = 15m;
let ApprovedPairs = datatable(CreatorRole:string, Workflow:string)
[
  "IAM-Onboarding", "approved-user-onboarding",
  "CiCdKeyRotator", "cicd-key-rotation",
  "BreakGlassKeyManager", "emergency-access"
];
let Creates = AWSCloudTrail
| where EventName == "CreateAccessKey" and isempty(ErrorCode)
| extend AccessKeyId = tostring(ResponseElements.accessKey.accessKeyId),
         Creator = coalesce(tostring(UserIdentity.userName), tostring(UserIdentity.arn)),
         CreatorRole = tostring(UserIdentity.sessionContext.sessionIssuer.userName),
         Workflow = tostring(parse_json(RequestParameters).workflow)
| project CreatedTime=TimeGenerated, AccessKeyId, Creator,
          CreateSourceIP=SourceIpAddress, CreatorRole, Workflow;
let FirstUses = AWSCloudTrail
| where EventName != "CreateAccessKey" and isempty(ErrorCode)
| extend AccessKeyId = tostring(UserIdentity.accessKeyId)
| where isnotempty(AccessKeyId)
| summarize arg_min(TimeGenerated, EventName, SourceIpAddress) by AccessKeyId
| project AccessKeyId, FirstUse=TimeGenerated,
          FirstUseEvent=EventName, FirstUseSourceIP=SourceIpAddress;
Creates
| join kind=inner FirstUses on AccessKeyId
| extend DeltaSeconds = datetime_diff("second", FirstUse, CreatedTime)
| where DeltaSeconds between (0 .. tolong(Window / 1s))
| where CreateSourceIP != FirstUseSourceIP
| join kind=leftanti ApprovedPairs on CreatorRole, Workflow
| project Creator, AccessKeyId, CreatedTime, FirstUse, FirstUseEvent,
          CreateSourceIP, FirstUseSourceIP, DeltaSeconds, CreatorRole, Workflow
| order by DeltaSeconds asc

Analyst reasoning and false positives

Approved user onboarding may create a key and test it from a separate onboarding host. CI/CD key rotation often creates a credential in one control-plane job and tests it from another runner. Emergency access workflows may move between hardened systems. Keep these cases only when the role/workflow pair, owner, ticket, source ranges, and time window all agree.

CrowdStrike CQL answer

Validation place: Falcon LogScale or Next-Gen SIEM Search after confirming the parser and exact field paths with one known CloudTrail event.

#event.dataset="aws.cloudtrail"
| case {
    eventName="CreateAccessKey"
      | accessKeyId := responseElements.accessKey.accessKeyId
      | phase := "create"
      | creator := coalesce([userIdentity.userName, userIdentity.arn])
      | creatorRole := userIdentity.sessionContext.sessionIssuer.userName
      | workflow := requestParameters.workflow;
    userIdentity.accessKeyId=*
      | accessKeyId := userIdentity.accessKeyId
      | phase := "use";
  }
| accessKeyId=*
| groupBy([accessKeyId], function=[
    collect([@timestamp, phase, eventName, sourceIPAddress,
             creator, creatorRole, workflow], limit=50)
  ], limit=max)
| sort(@timestamp, order=asc)
| createTime := array:first("@timestamp", where=[phase="create"])
| firstUseTime := array:first("@timestamp", where=[phase="use"])
| createSource := array:first(sourceIPAddress, where=[phase="create"])
| firstUseSource := array:first(sourceIPAddress, where=[phase="use"])
| creator := array:first(creator, where=[phase="create"])
| creatorRole := array:first(creatorRole, where=[phase="create"])
| workflow := array:first(workflow, where=[phase="create"])
| deltaSeconds := (firstUseTime-createTime)/1000
| test(deltaSeconds>=0 and deltaSeconds<=900)
| test(createSource!=firstUseSource)
| not((creatorRole="IAM-Onboarding" and workflow="approved-user-onboarding")
   or (creatorRole="CiCdKeyRotator" and workflow="cicd-key-rotation")
   or (creatorRole="BreakGlassKeyManager" and workflow="emergency-access"))
| table([creator, accessKeyId, createTime, firstUseTime,
         createSource, firstUseSource, deltaSeconds,
         creatorRole, workflow])

Analyst reasoning and false positives

The same three expected workflows can resemble the alert: approved onboarding, CI/CD rotation, and emergency access. Field names, array functions, and parser output vary by Falcon tenant, so first test the field map on known events. A broad role-only exclusion would hide unrelated key creation by a compromised session.

Analyst reasoning and false positives

A Sigma hit means a new credential crossed sources quickly; it is not proof of compromise. Confirm who requested the key, the target IAM user, CloudTrail session identity, source ownership, first API action, change ticket, and later activity. The false positives are approved onboarding, CI/CD rotation, and emergency access—but only with matching governance evidence.

Week 23 · Day 5

Simple Python correlation

Page 6 / 8

What should be done

Save the Day 1 lines as events.jsonl. The script remembers each new key, then evaluates the first later event that uses that key. It prints creator, key, API, source, and time delta for non-allowlisted matches.

Easy reading plan

  1. Read one JSON object per line.
  2. Store CreateAccessKey by returned key ID.
  3. On first use, compare time and source.
  4. Skip exact role/workflow pairs.
  5. Print the two expected alerts.

Python answer

import json
from datetime import datetime

WINDOW_SECONDS = 15 * 60
APPROVED = {
    ("IAM-Onboarding", "approved-user-onboarding"),
    ("CiCdKeyRotator", "cicd-key-rotation"),
    ("BreakGlassKeyManager", "emergency-access"),
}


def time_of(event):
    return datetime.fromisoformat(event["eventTime"].replace("Z", "+00:00"))


creates = {}
first_use_seen = set()

with open("events.jsonl", encoding="utf-8") as file:
    for line in file:
        event = json.loads(line)

        if event["eventName"] == "CreateAccessKey":
            key = event["responseElements"]["accessKey"]["accessKeyId"]
            creates[key] = event
            continue

        key = event.get("userIdentity", {}).get("accessKeyId")
        if not key or key not in creates or key in first_use_seen:
            continue

        first_use_seen.add(key)
        created = creates[key]
        delta = int((time_of(event) - time_of(created)).total_seconds())
        role = created["lab"].get("creatorRole", "")
        workflow = created["lab"].get("workflow", "")
        approved = (role, workflow) in APPROVED
        new_source = event["sourceIPAddress"] != created["sourceIPAddress"]

        if 0 <= delta <= WINDOW_SECONDS and new_source and not approved:
            creator = created["userIdentity"].get("userName", created["userIdentity"]["arn"])
            print(f"{creator} | {key} | {event['eventName']} | "
                  f"{event['sourceIPAddress']} | {delta}s")

Go answer

package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "sort"
    "time"
)

type Event struct {
    EventTime       string `json:"eventTime"`
    EventName       string `json:"eventName"`
    SourceIPAddress string `json:"sourceIPAddress"`
    UserIdentity struct {
        AccessKeyID string `json:"accessKeyId"`
    } `json:"userIdentity"`
    ResponseElements struct {
        AccessKey struct {
            AccessKeyID string `json:"accessKeyId"`
        } `json:"accessKey"`
    } `json:"responseElements"`
}

type Step struct {
    When   time.Time
    Phase  string
    Name   string
    Source string
}

func main() {
    file, err := os.Open("events.jsonl")
    if err != nil {
        fmt.Println("open:", err)
        return
    }
    defer file.Close()

    timelines := map[string][]Step{}
    scanner := bufio.NewScanner(file)

    for scanner.Scan() {
        var event Event
        if json.Unmarshal(scanner.Bytes(), &event) != nil {
            continue
        }

        key := event.UserIdentity.AccessKeyID
        phase := "USE"
        if event.EventName == "CreateAccessKey" {
            key = event.ResponseElements.AccessKey.AccessKeyID
            phase = "CREATE"
        }
        if key == "" {
            continue
        }

        when, err := time.Parse(time.RFC3339, event.EventTime)
        if err != nil {
            continue
        }
        timelines[key] = append(timelines[key], Step{
            When: when, Phase: phase, Name: event.EventName,
            Source: event.SourceIPAddress,
        })
    }

    keys := make([]string, 0, len(timelines))
    for key := range timelines {
        keys = append(keys, key)
    }
    sort.Strings(keys)

    for _, key := range keys {
        steps := timelines[key]
        sort.Slice(steps, func(i, j int) bool {
            return steps[i].When.Before(steps[j].When)
        })
        fmt.Println(key)
        for _, step := range steps {
            fmt.Printf("  %s | %-6s | %-18s | %s\n",
                step.When.Format("15:04:05"), step.Phase,
                step.Name, step.Source)
        }
    }
}

Analyst note

The Go program intentionally builds timelines instead of deciding alerts. It makes ordering errors, multiple uses, and missing creations visible. Day 7 applies the 15-minute, new-source, and allowlist decisions.
