Week 26 Â· Day 2

KQL behavior query + tuning

Page 3 / 8

Build it in five checks

  1. Normalize source tables to the lab fields.
  2. Group events by a stable investigation key.
  3. Count the five required stages.
  4. Require three beacon events and a 30-minute span.
  5. Exclude only evidence-backed approved workflows.

Important production note

CaseId is provided by this synthetic lab. In production, create a chain key from device, user, process lineage, paths, task details, and a bounded time window. Never assume unrelated events belong together merely because they share a host.

Do not copy the synthetic ApprovedWorkflow flag into production. Build it from owner, signer, path, account, ticket, scope, and expiry evidence.

KQL answer

Target: a normalized Week26Events table. Validate field names and joins before production use.

let HuntWindow = 30m;
Week26Events
| summarize
    StartTime = min(Timestamp),
    EndTime = max(Timestamp),
    OfficePowerShell = countif(
        EventType == "process"
        and ParentName in~ ("WINWORD.EXE", "EXCEL.EXE", "POWERPNT.EXE")
        and ProcessName in~ ("powershell.exe", "pwsh.exe")
    ),
    PowerShellDownloads = countif(
        EventType == "network" and Action == "download"
        and ProcessName in~ ("powershell.exe", "pwsh.exe")
    ),
    WritableFiles = countif(
        EventType == "file"
        and (Path has "\\AppData\\" or Path has "\\Temp\\" or Path has "\\Downloads\\")
    ),
    PowerShellTasks = countif(
        EventType == "scheduled_task" and Action == "create"
        and (TaskCommand has "powershell.exe" or TaskCommand has "pwsh.exe")
    ),
    BeaconEvents = countif(EventType == "network" and Action == "beacon"),
    BeaconDestinations = make_set_if(Destination, EventType == "network" and Action == "beacon"),
    Paths = make_set_if(Path, isnotempty(Path)),
    ApprovedEvidence = countif(ApprovedWorkflow == true)
  by CaseId, Device, User
| where EndTime - StartTime <= HuntWindow
| where OfficePowerShell > 0
    and PowerShellDownloads > 0
    and WritableFiles > 0
    and PowerShellTasks > 0
    and BeaconEvents >= 3
| where ApprovedEvidence == 0
| project StartTime, EndTime, CaseId, Device, User,
          BeaconDestinations, Paths, BeaconEvents
| order by StartTime asc

Analyst reasoning

False positives: macro-enabled business tools, sanctioned deployment scripts, approved scheduled updaters, and monitoring beacons. The query avoids broad allowlists by requiring explicit approval evidence and still asks the analyst to inspect signer, path, user, task owner, destination, prevalence, ticket, and expiry. A five-stage match raises confidence; it does not by itself prove a phishing compromise.

Python answer

import json
from collections import Counter, defaultdict
from datetime import datetime

OFFICE = {"winword.exe", "excel.exe", "powerpnt.exe"}
SHELLS = {"powershell.exe", "pwsh.exe"}


def name(path):
    return path.replace("\\", "/").split("/")[-1].lower()


def is_writable(path):
    text = path.lower()
    return "\\appdata\\" in text or "\\temp\\" in text or "\\downloads\\" in text


def is_incident(rows):
    rows.sort(key=lambda event: event["timestamp"])
    office_shell = any(
        event["event_type"] == "process"
        and name(event["parent"]) in OFFICE
        and name(event["process"]) in SHELLS
        for event in rows
    )
    download = any(
        event["event_type"] == "network"
        and event["action"] == "download"
        and name(event["process"]) in SHELLS
        for event in rows
    )
    writable_file = any(
        event["event_type"] == "file" and is_writable(event["path"])
        for event in rows
    )
    task = any(
        event["event_type"] == "scheduled_task"
        and event["action"] == "create"
        and any(shell in event["task_command"].lower() for shell in SHELLS)
        for event in rows
    )
    destinations = Counter(
        event["destination"] for event in rows
        if event["event_type"] == "network" and event["action"] == "beacon"
    )
    repeated_beacon = bool(destinations) and max(destinations.values()) >= 3
    approved = any(event["approved_workflow"] for event in rows)
    start = datetime.fromisoformat(rows[0]["timestamp"].replace("Z", "+00:00"))
    end = datetime.fromisoformat(rows[-1]["timestamp"].replace("Z", "+00:00"))
    short_window = (end - start).total_seconds() <= 30 * 60
    return all([office_shell, download, writable_file, task, repeated_beacon, short_window]) and not approved


by_case = defaultdict(list)
with open("events.jsonl", encoding="utf-8") as source:
    for line in source:
        event = json.loads(line)
        by_case[event["case_id"]].append(event)

for case_id, rows in sorted(by_case.items()):
    if is_incident(rows):
        stages = " â†’ ".join(event["action"] for event in rows)
        print(f"{case_id} | {rows[0]['device']} | {stages}")

Analyst reasoning

False positives: approved business macros, software deployment, updater tasks, and monitoring heartbeats. The program keeps the logic readable, but its synthetic case_id is a teaching shortcut. A production join must use device, process identity, user, path, task metadata, destination, and bounded timestamps; otherwise unrelated events could be joined into a false story.

Go answer

package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "sort"
    "strings"
    "time"
)

type Event struct {
    CaseID   string `json:"case_id"`
    Time     string `json:"timestamp"`
    Type     string `json:"event_type"`
    Device   string `json:"device"`
    Action   string `json:"action"`
    Process  string `json:"process"`
    Parent   string `json:"parent"`
    Path     string `json:"path"`
    TaskCmd  string `json:"task_command"`
    Dest     string `json:"destination"`
    Approved bool   `json:"approved_workflow"`
}

func name(path string) string {
    parts := strings.Split(strings.ReplaceAll(path, "\\", "/"), "/")
    return strings.ToLower(parts[len(parts)-1])
}

func has(items map[string]bool, value string) bool { return items[name(value)] }

func incident(rows []Event) bool {
    office := map[string]bool{"winword.exe": true, "excel.exe": true, "powerpnt.exe": true}
    shells := map[string]bool{"powershell.exe": true, "pwsh.exe": true}
    officeShell, download, writable, task, approved := false, false, false, false, false
    beacons := map[string]int{}
    for _, e := range rows {
        officeShell = officeShell || (e.Type == "process" && has(office, e.Parent) && has(shells, e.Process))
        download = download || (e.Type == "network" && e.Action == "download" && has(shells, e.Process))
        p := strings.ToLower(e.Path)
        writable = writable || (e.Type == "file" && (strings.Contains(p, "\\appdata\\") || strings.Contains(p, "\\temp\\") || strings.Contains(p, "\\downloads\\")))
        cmd := strings.ToLower(e.TaskCmd)
        task = task || (e.Type == "scheduled_task" && e.Action == "create" && (strings.Contains(cmd, "powershell.exe") || strings.Contains(cmd, "pwsh.exe")))
        if e.Type == "network" && e.Action == "beacon" { beacons[e.Dest]++ }
        approved = approved || e.Approved
    }
    repeated := false
    for _, count := range beacons { if count >= 3 { repeated = true } }
    start, _ := time.Parse(time.RFC3339, rows[0].Time)
    end, _ := time.Parse(time.RFC3339, rows[len(rows)-1].Time)
    return officeShell && download && writable && task && repeated && end.Sub(start) <= 30*time.Minute && !approved
}

func main() {
    file, _ := os.Open("events.jsonl")
    defer file.Close()
    byCase := map[string][]Event{}
    scanner := bufio.NewScanner(file)
    for scanner.Scan() {
        var e Event
        json.Unmarshal(scanner.Bytes(), &e)
        byCase[e.CaseID] = append(byCase[e.CaseID], e)
    }
    ids := make([]string, 0, len(byCase))
    for id := range byCase { ids = append(ids, id) }
    sort.Strings(ids)
    for _, id := range ids {
        rows := byCase[id]
        sort.Slice(rows, func(i, j int) bool { return rows[i].Time < rows[j].Time })
        if incident(rows) {
            parts := make([]string, 0, len(rows))
            for _, e := range rows { parts = append(parts, e.Action) }
            fmt.Printf("%s | %s | %s\n", id, rows[0].Device, strings.Join(parts, " â†’ "))
        }
    }
}

Analyst reasoning

False positives: sanctioned scripts and task-based updaters can create valid multi-event timelines, while monitoring agents can look periodic. Keep approval evidence narrow and time-bounded. The Go example is intentionally small: a production streamer would also handle malformed JSON, scanner limits, late events, state expiry, duplicates, clock skew, and persistent state.
