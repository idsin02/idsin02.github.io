Week 19 Â· Day 2

KQL behavior query + tuning

Page 3 / 8

Build it in four steps

  1. Keep Linux process creation.
  2. Match known web-service parents.
  3. Match shell or scripting children.
  4. Require Java to look like an app server, then suppress only a narrowly approved maintenance path.

Broad behavior first; precise context second. Do not filter every bash launched by a service account.

Tuning notes

  • Maintenance scripts: allowlist a signed path plus service identity, not a vague word.
  • Application hooks: confirm the exact parent command and child path.
  • Image builds: separate build-time containers from runtime workloads.
  • Diagnostics: use a short expiry and ticket reference for exceptions.

KQL answer

Validation place: Microsoft Defender Advanced Hunting. Confirm the Linux field coverage on one known event before scheduling the query.

let WebParents = dynamic(["nginx", "apache", "apache2", "httpd", "php-fpm"]);
let ScriptChildren = dynamic(["sh", "bash", "dash", "zsh", "python", "python3", "perl", "ruby", "php", "node"]);
DeviceProcessEvents
| where OSPlatform =~ "Linux"
| extend ParentName = tolower(InitiatingProcessFileName),
         ChildName = tolower(FileName)
| where ParentName in~ (WebParents)
    or (ParentName == "java" and InitiatingProcessCommandLine has_any
        ("org.apache.catalina", "jetty", "wildfly", "jboss"))
| where ChildName in~ (ScriptChildren)
| where not(InitiatingProcessCommandLine has "maintenance-runner"
    and ProcessCommandLine startswith "/bin/bash /opt/approved/")
| project Timestamp, DeviceName, AccountName,
          ParentName, InitiatingProcessCommandLine,
          ChildName, ProcessCommandLine,
          ProcessId, InitiatingProcessId,
          AdditionalFields
| order by Timestamp desc

Analyst reasoning

The query finds high-signal relationships, not confirmed compromise. False positives can include approved maintenance scripts, application lifecycle hooks, container image build steps, and temporary diagnostics. Validate ownership, image digest, deployment time, UID, command arguments, and whether the child made network or file changes. An exception should bind multiple stable facts and have an expiry date.

Week 19 Â· Day 5

Normalize fixtures with Python

Page 6 / 8

What this script does

  1. Reads one JSON object per line.
  2. Normalizes parent and child executable basenames.
  3. Preserves UID, workload, time, PID, and PPID.
  4. Evaluates the same parent â†’ child behavior.
  5. Prints one small normalized JSON object per event.

Run it

python3 normalize.py events.jsonl

Expected summary: TP=3 TN=3 mistakes=0. The script exits with code 1 if any fixture is classified incorrectly.

Simple code is a feature. Readable normalization lets an analyst spot bad mappings quickly.

Python answer

import json
import os
import sys

WEB = {"nginx", "apache", "apache2", "httpd", "php-fpm"}
SCRIPT = {"sh", "bash", "dash", "zsh", "python", "python3", "perl", "ruby", "php", "node"}
JAVA_SERVERS = ("org.apache.catalina", "jetty", "wildfly", "jboss")


def base(path):
    return os.path.basename(path or "").lower()


def normalize(event):
    return {
        "timestamp": event.get("timestamp"),
        "host": event.get("node"),
        "container": event.get("container_id"),
        "uid": event.get("uid"),
        "parent": base(event.get("parent_exe")),
        "parent_command": event.get("parent_cmdline", ""),
        "child": base(event.get("exe")),
        "child_command": event.get("command_line", ""),
        "pid": event.get("pid"),
        "ppid": event.get("ppid"),
        "expected": event.get("expected"),
    }


def risky(event):
    parent_ok = event["parent"] in WEB
    if event["parent"] == "java":
        text = event["parent_command"].lower()
        parent_ok = any(name in text for name in JAVA_SERVERS)
    return parent_ok and event["child"] in SCRIPT


counts = {"TP": 0, "TN": 0}
mistakes = 0
with open(sys.argv[1], encoding="utf-8") as source:
    for line in source:
        event = normalize(json.loads(line))
        detected = risky(event)
        counts[event["expected"]] += 1
        mistakes += detected != (event["expected"] == "TP")
        event["detected"] = detected
        print(json.dumps(event, separators=(",", ":")))

print(f"TP={counts['TP']} TN={counts['TN']} mistakes={mistakes}")
sys.exit(1 if mistakes else 0)

Analyst reasoning

Normalization errors can look like detection failures. False positives can come from maintenance scripts, application hooks, image build steps, and diagnostics; the Python sample keeps the detector broad so those cases remain visible for deliberate tuning. In production, log the normalization version and the original record identifier.

Go answer

package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "path/filepath"
    "strings"
)

type Event struct {
    ID            string `json:"id"`
    Node          string `json:"node"`
    ParentExe     string `json:"parent_exe"`
    ParentCommand string `json:"parent_cmdline"`
    Exe           string `json:"exe"`
    Command       string `json:"command_line"`
    PID           int    `json:"pid"`
    PPID          int    `json:"ppid"`
}

func base(path string) string {
    return strings.ToLower(filepath.Base(path))
}

func has(items []string, value string) bool {
    for _, item := range items {
        if item == value {
            return true
        }
    }
    return false
}

func risky(e Event) bool {
    parent := base(e.ParentExe)
    child := base(e.Exe)
    web := has([]string{"nginx", "apache", "apache2", "httpd", "php-fpm"}, parent)
    if parent == "java" {
        text := strings.ToLower(e.ParentCommand)
        web = strings.Contains(text, "org.apache.catalina") ||
            strings.Contains(text, "jetty") ||
            strings.Contains(text, "wildfly") ||
            strings.Contains(text, "jboss")
    }
    script := has([]string{"sh", "bash", "dash", "zsh", "python", "python3", "perl", "ruby", "php", "node"}, child)
    return web && script
}

func main() {
    file, err := os.Open(os.Args[1])
    if err != nil {
        panic(err)
    }
    defer file.Close()

    scanner := bufio.NewScanner(file)
    for scanner.Scan() {
        var event Event
        if json.Unmarshal(scanner.Bytes(), &event) != nil {
            continue
        }
        if risky(event) {
            fmt.Printf("%s %s %s[%d] â†’ %s[%d] %s\n",
                event.ID, event.Node, base(event.ParentExe), event.PPID,
                base(event.Exe), event.PID, event.Command)
        }
    }
    if err := scanner.Err(); err != nil {
        panic(err)
    }
}
