Day 2 · KQL
Build the behavior query
Page 3 / 8

  

    
Think in three sets
  1. Baseline: successful type-10 logons from the previous 30 days.
  2. Current: successful type-10 logons from the last day.
  3. Result: sources missing from both the user's and device's baseline.

    
Tuning before alerting
  • Use exact source IPs for jump servers and support systems.
  • Use a reviewed CIDR only for a controlled VPN egress range.
  • Give disaster-recovery exceptions an owner and expiry.
  • Keep “new source” as evidence, not proof of compromise.

    
KQL answer
The query uses Windows SecurityEvent 4624 and compares the current IP with separate user and device source sets.
let CurrentStart = ago(1d);
let BaselineStart = ago(30d);
let RdpSuccess = SecurityEvent
| where EventID == 4624 and LogonType == 10
| where isnotempty(IpAddress) and IpAddress !in ("-", "127.0.0.1", "::1")
| project TimeGenerated, Account, Computer, IpAddress;
let UserBaseline = RdpSuccess
| where TimeGenerated between (BaselineStart .. CurrentStart)
| summarize UserSources = make_set(IpAddress, 1024) by Account;
let DeviceBaseline = RdpSuccess
| where TimeGenerated between (BaselineStart .. CurrentStart)
| summarize DeviceSources = make_set(IpAddress, 1024) by Computer;
RdpSuccess
| where TimeGenerated > CurrentStart
| join kind=leftouter UserBaseline on Account
| join kind=leftouter DeviceBaseline on Computer
| extend NewForUser = coalesce(set_has_element(UserSources, IpAddress), false) == false
| extend NewForDevice = coalesce(set_has_element(DeviceSources, IpAddress), false) == false
| extend ApprovedException =
    IpAddress in ("10.20.4.15", "10.20.9.20", "10.99.0.10")
    or ipv4_is_in_range(IpAddress, "10.70.0.0/16")
| where NewForUser and NewForDevice and not(ApprovedException)
| project TimeGenerated, Account, Computer, IpAddress, NewForUser, NewForDevice
| order by TimeGenerated desc

Python answer
Save the downloaded CSV and JSONL beside this script. The code uses two sets and prints only new, non-exempt RDP events.
import csv
import ipaddress
import json

known_user_sources = set()
known_device_sources = set()

with open("week8_baseline.csv", newline="", encoding="utf-8") as handle:
    for row in csv.DictReader(handle):
        if row["kind"] == "user":
            known_user_sources.add((row["user"].lower(), row["source_ip"]))
        if row["kind"] == "device":
            known_device_sources.add((row["target_device"].lower(), row["source_ip"]))

approved_devices = {"jump-01", "helpdesk-02", "dr-jump-01"}
approved_vpn = ipaddress.ip_network("10.70.0.0/16")

def approved_exception(event):
    source_device = event["source_device"].lower()
    source_ip = ipaddress.ip_address(event["source_ip"])
    return (
        event.get("approved_support", False)
        or source_device in approved_devices
        or source_ip in approved_vpn
    )

def is_new_rdp(event):
    if not event["success"] or event["logon_type"] != 10:
        return False
    user_pair = (event["user"].lower(), event["source_ip"])
    device_pair = (event["target_device"].lower(), event["source_ip"])
    new_for_user = user_pair not in known_user_sources
    new_for_device = device_pair not in known_device_sources
    return new_for_user and new_for_device and not approved_exception(event)

with open("week8_events.jsonl", encoding="utf-8") as handle:
    for line in handle:
        event = json.loads(line)
        if is_new_rdp(event):
            print(event["id"], event["user"], "→", event["target_device"])

Day 6 · Go
Compare against a local allowlist map
Page 7 / 8

  

    
Why a map?
A Go map gives a fast yes/no check. Join the account or target device with the source IP to make a simple key.

    
Beginner goal
Focus on the comparison, not file parsing. The sample keeps four current events in code: two alerts and two approved exceptions.

    
Go answer
package main

import (
    "fmt"
    "strings"
)

type Event struct {
    ID, User, Target, SourceIP, SourceDevice string
    LogonType int
    Success, ApprovedSupport bool
}

func key(left, sourceIP string) string {
    return strings.ToLower(left) + "|" + sourceIP
}

func main() {
    knownUser := map[string]bool{
        key(`CONTOSO\alice`, "10.10.5.44"): true,
        key(`CONTOSO\alice`, "10.20.4.15"): true,
    }
    knownDevice := map[string]bool{
        key("WS-FIN-07", "10.10.5.44"): true,
        key("WS-FIN-07", "10.20.4.15"): true,
    }
    approvedDevice := map[string]bool{
        "jump-01": true, "helpdesk-02": true, "dr-jump-01": true,
    }
    events := []Event{
        {"R1", `CONTOSO\alice`, "WS-FIN-07", "203.0.113.24", "UNKNOWN", 10, true, false},
        {"R2", `CONTOSO\bob`, "WS-HR-02", "198.51.100.77", "LAPTOP-441", 10, true, false},
        {"R3", `CONTOSO\alice`, "WS-FIN-07", "10.20.4.15", "JUMP-01", 10, true, false},
        {"R4", `CONTOSO\dave`, "WS-MKT-03", "10.20.9.20", "HELPDESK-02", 10, true, true},
    }
    for _, event := range events {
        newUser := !knownUser[key(event.User, event.SourceIP)]
        newDevice := !knownDevice[key(event.Target, event.SourceIP)]
        excepted := event.ApprovedSupport || approvedDevice[strings.ToLower(event.SourceDevice)]
        if event.Success && event.LogonType == 10 && newUser && newDevice && !excepted {
            fmt.Println(event.ID, event.User, "→", event.Target)
        }
    }
}

Day 7 · Capstone
Detect the new source, keep exceptions
Page 8 / 8

  

    
Analyst reasoning
  • Help-desk support: require a ticket, named support system, owner, and short expiry.
  • Jump servers: allow exact managed sources, not broad internal ranges.
  • VPN changes: update the approved egress range only after network-owner confirmation.
  • Disaster recovery: tie exceptions to planned exercises and remove them afterward.

    
Investigate an alert
  1. Confirm the source IP/device and whether it is expected for the user.
  2. Check the target's sensitivity and the account's privilege.
  3. Review nearby authentication failures, VPN activity, and other remote logons.
  4. Contact the user or system owner through an approved channel.
  5. Escalate only after context supports compromise.
