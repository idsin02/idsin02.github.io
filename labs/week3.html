Week 3 · Day 2

KQL behavior query + tuning

Page 3 / 8

Easy plan

  1. Select the four download tools.
  2. Join network events by device and process ID.
  3. Join file events the same way.
  4. Keep user-writable destinations.
  5. Require all evidence within five minutes.

Narrow tuning

The example suppresses only a winget.exe parent using the exact corporate package domain. It does not allow every PowerShell download or every corporate user.

Confirm the repository owner and review the exception regularly.

KQL answer

Use Microsoft Defender Advanced Hunting after checking the three table and field names in your tenant.

let ApprovedPackageDomains = dynamic(["packages.corp.example"]);
let Downloaders =
    DeviceProcessEvents
    | where FileName in~ ("powershell.exe", "pwsh.exe", "certutil.exe", "curl.exe", "bitsadmin.exe")
    | where ProcessCommandLine matches regex @"(?i)https?://"
    | where ProcessCommandLine has_any (@"\Downloads\", @"\Desktop\", @"\AppData\Local\Temp\")
    | where not(
        InitiatingProcessFileName =~ "winget.exe"
        and ProcessCommandLine has_any (ApprovedPackageDomains)
    )
    | project DeviceId, DeviceName, ProcessId, ProcTime=Timestamp,
              Tool=FileName, Parent=InitiatingProcessFileName,
              CommandLine=ProcessCommandLine;
let Network =
    DeviceNetworkEvents
    | where InitiatingProcessFileName in~ ("powershell.exe", "pwsh.exe", "certutil.exe", "curl.exe", "bitsadmin.exe")
    | project DeviceId, ProcessId=InitiatingProcessId,
              NetTime=Timestamp, Domain=RemoteUrl, RemoteIP;
let Files =
    DeviceFileEvents
    | where ActionType in ("FileCreated", "FileRenamed")
    | extend TargetPath=strcat(FolderPath, @"\", FileName)
    | where TargetPath matches regex @"(?i)\\Users\\[^\\]+\\(Downloads|Desktop|AppData\\Local\\Temp)\\"
    | project DeviceId, ProcessId=InitiatingProcessId,
              FileTime=Timestamp, TargetPath, SHA256;
Downloaders
| join kind=inner Network on DeviceId, ProcessId
| join kind=inner Files on DeviceId, ProcessId
| where NetTime between (ProcTime .. ProcTime + 5m)
| where FileTime between (ProcTime .. ProcTime + 5m)
| project ProcTime, DeviceName, Tool, Parent, Domain, RemoteIP,
          TargetPath, SHA256, CommandLine
| order by ProcTime desc

Analyst notes

Likely false positives: package managers, browser helpers that hand a URL to a command-line utility, enterprise installers, and managed update agents. Validate the parent process, signer, domain owner, destination, hash reputation, user intent, and nearby process activity before closing an alert.

Python answer

import json
from pathlib import PureWindowsPath

TOOLS = {"powershell.exe", "pwsh.exe", "certutil.exe", "curl.exe", "bitsadmin.exe"}
USER_DIRS = ("\\downloads\\", "\\desktop\\", "\\appdata\\local\\temp\\")


def name(path):
    return PureWindowsPath(path).name.lower()


def is_download(process, file_event):
    tool_ok = process["tool"].lower() in TOOLS
    url_ok = process["url"].lower().startswith(("http://", "https://"))
    path_ok = "\\users\\" in file_event["target_path"].lower()
    path_ok = path_ok and any(x in file_event["target_path"].lower() for x in USER_DIRS)
    approved = name(process["parent_image"]) == "winget.exe"
    approved = approved and process["domain"].lower() == "packages.corp.example"
    return tool_ok and url_ok and path_ok and not approved


processes = {}
with open("processes.jsonl", encoding="utf-8") as source:
    for line in source:
        process = json.loads(line)
        processes[process["process_id"]] = process

with open("files.jsonl", encoding="utf-8") as source:
    for line in source:
        file_event = json.loads(line)
        process = processes.get(file_event["process_id"])
        if process and is_download(process, file_event):
            print(f'{process["process_id"]} → {file_event["target_path"]} → {file_event["sha256"][:12]}')

Analyst notes

Likely false positives: package managers, browser helpers, enterprise installers, and update agents. The sample filter uses an exact parent + domain exception; production code should also check signing, ownership, expiry, and expected device scope.

Week 3 · Day 6

Summarize extensions with Go + scan with YARA

Page 7 / 8

Go task

Read download_events.jsonl, apply the same easy detector, take the extension from each matching destination, and count it.

Expected: .bin: 1, .dat: 1, .exe: 1, .ps1: 1.

Why YARA is separate

The process rule finds transfer behavior. YARA scans bytes after a file exists. The training rule below matches a harmless synthetic marker and domain; it is not a malware family rule.

Go answer

package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "path/filepath"
    "sort"
    "strings"
)

type Event struct {
    ProcessID  int    `json:"process_id"`
    Parent     string `json:"parent_image"`
    Tool       string `json:"tool"`
    URL        string `json:"url"`
    Domain     string `json:"domain"`
    TargetPath string `json:"target_path"`
}

func base(path string) string {
    return strings.ToLower(filepath.Base(strings.ReplaceAll(path, "\\", "/")))
}

func isDownload(e Event) bool {
    tools := map[string]bool{"powershell.exe": true, "pwsh.exe": true, "certutil.exe": true, "curl.exe": true, "bitsadmin.exe": true}
    path := strings.ToLower(e.TargetPath)
    userDir := strings.Contains(path, "\\users\\") &&
        (strings.Contains(path, "\\downloads\\") || strings.Contains(path, "\\desktop\\") || strings.Contains(path, "\\appdata\\local\\temp\\"))
    approved := base(e.Parent) == "winget.exe" && strings.EqualFold(e.Domain, "packages.corp.example")
    return tools[strings.ToLower(e.Tool)] && strings.HasPrefix(strings.ToLower(e.URL), "http") && userDir && !approved
}

func main() {
    file, err := os.Open("download_events.jsonl")
    if err != nil { panic(err) }
    defer file.Close()

    counts := map[string]int{}
    scan := bufio.NewScanner(file)
    for scan.Scan() {
        var event Event
        if json.Unmarshal(scan.Bytes(), &event) == nil && isDownload(event) {
            clean := strings.ReplaceAll(event.TargetPath, "\\", "/")
            counts[strings.ToLower(filepath.Ext(clean))]++
        }
    }
    if err := scan.Err(); err != nil { panic(err) }

    extensions := make([]string, 0, len(counts))
    for extension := range counts { extensions = append(extensions, extension) }
    sort.Strings(extensions)
    for _, extension := range extensions { fmt.Printf("%s: %d\n", extension, counts[extension]) }
}
