Week 21 · Day 2

KQL behavior query + tuning

Page 3 / 8

What should be done

  1. Split creation and privilege-change records.
  2. Join on host, target user, source, ticket, and context.
  3. Require creation first and a ten-minute window.
  4. Describe the privilege change in one field.
  5. Remove only exact, owned provisioning paths.

Why the tuning is narrow

The exception combines workflow source, actor, command path, and ticket prefix. A trusted actor or source alone is too broad and could hide manual misuse.

Replace sample actors, paths, groups, and ticket rules with your CMDB and provisioning inventory. Give every exception an owner and review date.

KQL answer

let Window = 10m;
let Creates = LinuxAuthAudit_CL
| where Action_s == "user_create"
| project CreatedAt=TimeGenerated, Host=tostring(Host_s), TargetUser=tostring(TargetUser_s),
          CreateActor=tostring(Actor_s), CreateCommand=tostring(Command_s),
          Source=tostring(Source_s), Ticket=tostring(Ticket_s),
          ApprovedContext=tostring(ApprovedContext_s);
let PrivilegeChanges = LinuxAuthAudit_CL
| where (Action_s == "group_add" and Group_s in~ ("sudo", "wheel"))
    or (Action_s == "sudoers_write" and SudoersFile_s startswith "/etc/sudoers")
| project ChangedAt=TimeGenerated, Host=tostring(Host_s), TargetUser=tostring(TargetUser_s),
          ChangeActor=tostring(Actor_s), PrivilegeAction=tostring(Action_s),
          PrivilegedGroup=tostring(Group_s), SudoersFile=tostring(SudoersFile_s),
          ChangeCommand=tostring(Command_s), Source=tostring(Source_s),
          Ticket=tostring(Ticket_s), ApprovedContext=tostring(ApprovedContext_s);
Creates
| join kind=inner PrivilegeChanges on Host, TargetUser, Source, Ticket, ApprovedContext
| where ChangedAt between (CreatedAt .. CreatedAt + Window)
| extend PrivilegeChange = case(
    PrivilegeAction == "group_add", strcat("group:", PrivilegedGroup),
    strcat("sudoers:", SudoersFile))
| where not(
    (Source == "automated_provisioning" and CreateActor == "provbot"
      and CreateCommand startswith "/opt/provision/new-user " and Ticket startswith "REQ-")
    or (Source == "cloud_init" and CreateActor == "root"
      and CreateCommand startswith "/usr/bin/cloud-init " and Ticket startswith "IMG-")
    or (Source == "configuration_management" and CreateActor == "ansible"
      and CreateCommand startswith "/usr/bin/ansible " and Ticket startswith "CHG-")
    or (Source == "break_glass_setup" and CreateActor == "secadmin"
      and ChangeCommand startswith "/usr/local/sbin/break-glass " and Ticket startswith "INC-")
)
| project CreatedAt, ChangedAt, Host, CreateActor, ChangeActor, TargetUser,
          PrivilegeChange, Source, Ticket, ApprovedContext, CreateCommand, ChangeCommand
| order by ChangedAt desc

Analyst reasoning

Likely false positives: automated provisioning, cloud-init, configuration management, and break-glass setup. Confirm the workflow owner, exact executable, source, ticket, requested target, intended group or sudoers scope, host build stage, expiry, and whether the recorded commands match the approved design.

CrowdStrike CQL answer

platform="Linux" action=/^(user_create|group_add|sudoers_write)$/
| selfJoinFilter([host, target_user],
    where=[
      {action="user_create"},
      {(action="group_add" group=/^(sudo|wheel)$/)
        OR (action="sudoers_write" sudoers_file=/^\/etc\/sudoers/)}
    ], prefilter=true)
| groupBy([host, target_user, source, ticket, approved_context],
    function=[min(@timestamp, as=created_at), max(@timestamp, as=changed_at),
              collect([actor, action, group, sudoers_file, command])])
| test(changed_at-created_at <= duration("10m"))
| case {
    source="automated_provisioning" actor="provbot" command=/^\/opt\/provision\// ticket=/^REQ-/ | drop();
    source="cloud_init" actor="root" command=/^\/usr\/bin\/cloud-init / ticket=/^IMG-/ | drop();
    source="configuration_management" actor="ansible" command=/^\/usr\/bin\/ansible / ticket=/^CHG-/ | drop();
    source="break_glass_setup" actor="secadmin" command=/^\/usr\/local\/sbin\/break-glass / ticket=/^INC-/ | drop();
    *
  }
| select([created_at, changed_at, host, actor, target_user, action,
          group, sudoers_file, source, ticket, approved_context, command])

Analyst reasoning

Likely false positives: automated provisioning, cloud-init, configuration management, and break-glass setup. Validate parser normalization, event order, host identity, target user, command, ticket, and approved context. Test selfJoinFilter, duration arithmetic, regular expressions, and collected multi-value fields against your LogScale version.

Why there is no YARA

This behavior lives in identity and configuration telemetry: account commands, group membership, audit records, and sudoers changes. YARA scans file content, so it would not add useful evidence here.

Correlation limit

This Sigma rule produces useful single-event signals; it does not prove the two-event sequence by itself. Use KQL, CQL, Python, Go, or a SIEM correlation rule to join creation and privilege events.

Analyst reasoning

Likely false positives: automated provisioning, cloud-init, configuration management, and break-glass setup. Before filtering, verify exact actor identity, executable path, command parameters, change ticket, workflow owner, target account, requested privilege, expiry, and the host's provisioning stage.

Week 21 · Day 5

Python: correlate creation and privilege records

Page 6 / 8

What should be done

Save Day 1 data as events.jsonl. Remember the latest user_create for each host and target. When a privileged group or sudoers record arrives, check the ten-minute window and the exact approved workflow.

Easy reading plan

  1. Read one JSON object per line.
  2. Sort by timestamp.
  3. Remember creation by host + target.
  4. Convert a qualifying grant to readable text.
  5. Skip exact approved workflows.
  6. Print the two cases needing review.

Python answer

import json
from datetime import datetime


def when(text):
    return datetime.fromisoformat(text.replace("Z", "+00:00"))


def privilege_change(event):
    if event["action"] == "group_add" and event["group"] in {"sudo", "wheel"}:
        return "group:" + event["group"]
    if event["action"] == "sudoers_write" and event["sudoers_file"].startswith("/etc/sudoers"):
        return "sudoers:" + event["sudoers_file"]
    return ""


def approved(event):
    checks = {
        "automated_provisioning": ("provbot", "/opt/provision/", "REQ-"),
        "cloud_init": ("root", "/usr/bin/cloud-init ", "IMG-"),
        "configuration_management": ("ansible", "/usr/bin/ansible ", "CHG-"),
        "break_glass_setup": ("secadmin", "/usr/local/sbin/break-glass ", "INC-"),
    }
    rule = checks.get(event["source"])
    return bool(rule and event["actor"] == rule[0]
                and event["command"].startswith(rule[1])
                and event["ticket"].startswith(rule[2]))


events = []
with open("events.jsonl", encoding="utf-8") as file:
    for line in file:
        events.append(json.loads(line))

events.sort(key=lambda event: event["timestamp"])
created = {}

for event in events:
    key = (event["host"], event["target_user"])
    if event["action"] == "user_create":
        created[key] = event
        continue

    change = privilege_change(event)
    first = created.get(key)
    if not change or not first:
        continue

    seconds = (when(event["timestamp"]) - when(first["timestamp"])).total_seconds()
    if 0 <= seconds <= 600 and not approved(event):
        print(event["case_id"], event["host"], event["actor"],
              event["target_user"], change)

Go answer

package main

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "strings"
    "time"
)

type Event struct {
    CaseID          string `json:"case_id"`
    Timestamp       string `json:"timestamp"`
    Host            string `json:"host"`
    Actor           string `json:"actor"`
    TargetUser      string `json:"target_user"`
    Action          string `json:"action"`
    Group           string `json:"group"`
    SudoersFile     string `json:"sudoers_file"`
    Command         string `json:"command"`
    Source          string `json:"source"`
    Ticket          string `json:"ticket"`
    ApprovedContext string `json:"approved_context"`
}

func change(e Event) string {
    if e.Action == "group_add" && (e.Group == "sudo" || e.Group == "wheel") {
        return "group:" + e.Group
    }
    if e.Action == "sudoers_write" && strings.HasPrefix(e.SudoersFile, "/etc/sudoers") {
        return "sudoers:" + e.SudoersFile
    }
    return ""
}

func approved(e Event) bool {
    rules := map[string][3]string{
        "automated_provisioning": {"provbot", "/opt/provision/", "REQ-"},
        "cloud_init": {"root", "/usr/bin/cloud-init ", "IMG-"},
        "configuration_management": {"ansible", "/usr/bin/ansible ", "CHG-"},
        "break_glass_setup": {"secadmin", "/usr/local/sbin/break-glass ", "INC-"},
    }
    rule, ok := rules[e.Source]
    return ok && e.Actor == rule[0] && strings.HasPrefix(e.Command, rule[1]) &&
        strings.HasPrefix(e.Ticket, rule[2])
}

func main() {
    file, err := os.Open("events.jsonl")
    if err != nil { panic(err) }
    defer file.Close()

    created := map[string]Event{}
    scanner := bufio.NewScanner(file)
    fmt.Println("HOST\tACTOR\tTARGET\tPRIVILEGE_CHANGE\tAPPROVED_CONTEXT\tDECISION")

    for scanner.Scan() {
        var event Event
        if err := json.Unmarshal(scanner.Bytes(), &event); err != nil { panic(err) }
        key := event.Host + "|" + event.TargetUser
        if event.Action == "user_create" {
            created[key] = event
            continue
        }

        first, found := created[key]
        privilege := change(event)
        if !found || privilege == "" { continue }
        start, _ := time.Parse(time.RFC3339, first.Timestamp)
        end, _ := time.Parse(time.RFC3339, event.Timestamp)
        if end.Sub(start) < 0 || end.Sub(start) > 10*time.Minute { continue }

        context, decision := "none", "REVIEW"
        if approved(event) {
            context, decision = event.ApprovedContext, "APPROVED"
        }
        fmt.Printf("%s\t%s\t%s\t%s\t%s\t%s\n",
            event.Host, event.Actor, event.TargetUser, privilege, context, decision)
    }
    if err := scanner.Err(); err != nil { panic(err) }
}
